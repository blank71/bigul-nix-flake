keyAlign :: Eq k => (s -> k) -> (v -> k) -> BiGUL s v -> (v -> s) -> BiGUL [s] [v]
keyAlign ks kv b c = [bigul|
  case
    normal [] [] exit []
      rearrV [] -> ()
         skip const ()
    normal (s : _) (v : _) | ks s = kv v exit (_ : _)
      rearrS (s : ss) -> (s, ss)
        rearrV (v : vs) -> (v, vs)
             b
          *  keyAlign ks kv b c
    adaptive (_ : _) []
      \ _ _ -> []
    adaptive ss (v : vs) | kv v `elem` map ks ss
      \ ss (v : _) -> uncurry (:) (extract (kv v) ss)
    adaptive _ (_ : _)
      \ ss (v : _) -> c v : ss
  |]
